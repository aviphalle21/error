// ─── Core booking completion (called once payment is confirmed Paid) ───────────
function completePaidBooking(PDO $pdo, int $user_id, string $table_number, string $payment_reference, array $user): array
{
    $pdo->beginTransaction();
    try {
        $paymentStmt = $pdo->prepare("
            SELECT p.payment_id, p.subscription_id, p.amount, p.payment_status,
                   p.created_at, us.plan_id, us.table_id, us.expiry_date, sp.duration_days
            FROM payments p
            JOIN user_subscriptions us ON p.subscription_id = us.subscription_id
            JOIN subscription_plans sp ON us.plan_id = sp.plan_id
            WHERE p.payment_reference = ? AND p.user_id = ?
            FOR UPDATE
        ");
        $paymentStmt->execute([$payment_reference, $user_id]);
        $payment = $paymentStmt->fetch();

        if (!$payment) {
            throw new Exception('Payment request not found.');
        }

        $secondsSinceCreated = $payment['created_at'] ? time() - strtotime($payment['created_at']) : 0;

        // Payment window expired → mark as Failed
        if ($payment['payment_status'] === 'Pending' && $secondsSinceCreated > PAYMENT_WINDOW_SECONDS) {
            $pdo->prepare("UPDATE payments SET payment_status = 'Failed', payment_date = NOW() WHERE payment_id = ?")->execute([$payment['payment_id']]);
            $pdo->prepare("UPDATE user_subscriptions SET payment_status = 'Failed', subscription_status = 'Inactive' WHERE subscription_id = ?")->execute([$payment['subscription_id']]);
            $pdo->commit();
            return [
                'completed' => false,
                'timed_out' => true,
                'failed'    => true,
                'redirect'  => 'payment.php?table_id=' . urlencode($table_number) . '&failed=1&reason=' . urlencode('Payment window expired. If any amount was deducted it will be automatically refunded within 24 hours.'),
                'message'   => 'Payment window expired.',
            ];
        }

        // Already Failed or Refunded
        if (in_array($payment['payment_status'], ['Failed', 'Refunded'])) {
            $pdo->commit();
            $reason = $payment['payment_status'] === 'Refunded'
                ? 'This payment was already refunded.'
                : 'Payment was not completed. If any amount was deducted it will be refunded within 24 hours.';
            return [
                'completed' => false,
                'failed'    => true,
                'redirect'  => 'payment.php?table_id=' . urlencode($table_number) . '&failed=1&reason=' . urlencode($reason),
                'message'   => $reason,
            ];
        }

        // Still Pending (user hasn't paid yet)
        if ($payment['payment_status'] !== 'Paid') {
            $pdo->commit();
            $windowLeft = PAYMENT_WINDOW_SECONDS - $secondsSinceCreated;
            return [
                'completed'    => false,
                'timed_out'    => false,
                'message'      => 'Waiting for payment confirmation. Please complete UPI payment and click "Ive Paid".',
                'seconds_left' => max(0, $windowLeft),
            ];
        }

        // Payment is Paid → create booking if not already done
        $activeStmt = $pdo->prepare("SELECT booking_id FROM bookings WHERE user_id = ? AND table_id = ? AND booking_status = 'Active'");
        $activeStmt->execute([$user_id, $payment['table_id']]);
        if (!$activeStmt->fetch()) {
            $tableStmt = $pdo->prepare("SELECT status FROM library_tables WHERE table_id = ? FOR UPDATE");
            $tableStmt->execute([$payment['table_id']]);
            $tableRow = $tableStmt->fetch();

            if (!$tableRow || $tableRow['status'] === 'Booked') {
                $pdo->rollBack();
                return [
                    'completed' => false,
                    'failed'    => true,
                    'redirect'  => 'payment.php?table_id=' . urlencode($table_number) . '&failed=1&reason=' . urlencode('This table was taken by someone else. Your payment of ₹' . $payment['amount'] . ' will be refunded within 24 hours.'),
                    'message'   => 'Table no longer available.',
                ];
            }

            // Activate subscription
            $pdo->prepare("UPDATE user_subscriptions SET payment_status = 'Paid', subscription_status = 'Active', start_date = CURDATE(), expiry_date = DATE_ADD(CURDATE(), INTERVAL ? DAY) WHERE subscription_id = ?")
                ->execute([$payment['duration_days'], $payment['subscription_id']]);

            // Create booking
            $bookingRef = 'BK-' . mt_rand(100000, 999999);
            $pdo->prepare("INSERT INTO bookings (user_id, table_id, start_date, expiry_date, booking_status, booking_reference, plan_price, booking_price) VALUES (?, ?, CURDATE(), DATE_ADD(CURDATE(), INTERVAL ? DAY), 'Active', ?, ?, ?)")
                ->execute([$user_id, $payment['table_id'], $payment['duration_days'], $bookingRef, $payment['amount'], $payment['amount']]);

            // Mark table booked
            $pdo->prepare("UPDATE library_tables SET status = 'Booked', current_user_id = ? WHERE table_id = ?")
                ->execute([$user_id, $payment['table_id']]);

            // Notification
            $pdo->prepare("INSERT INTO system_notifications (type, title, message) VALUES ('payment', 'New Payment & Booking', ?)")
                ->execute([$user['full_name'] . " paid ₹" . $payment['amount'] . " and booked Table T-{$table_number}."]);

            // Email
            @mail(
                $user['email'],
                'Booking Confirmed – Saraswati Abhyasika',
                "Hello {$user['full_name']},\n\nYour booking for Table T-{$table_number} is confirmed!\n\nBooking Reference: {$bookingRef}\nAmount Paid: ₹{$payment['amount']}\n\nThank you for choosing Saraswati Abhyasika!",
                "From: noreply@saraswatiabhyasika.com\r\nX-Mailer: PHP/" . phpversion()
            );

            // SMS log
            file_put_contents(
                'notification_logs.txt',
                '[' . date('Y-m-d H:i:s') . "] SMS queued for {$user['phone']}: Booking T-{$table_number} confirmed. Paid ₹{$payment['amount']}. Ref: {$bookingRef}\n",
                FILE_APPEND
            );
        }

        $pdo->commit();
        return [
            'completed' => true,
            'redirect'  => 'payment.php?table_id=' . urlencode($table_number) . '&success=1',
        ];
    } catch (Exception $e) {
        $pdo->rollBack();
        throw $e;
    }
}

// ─── AJAX endpoint ─────────────────────────────────────────────────────────────
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
    try {

        // ── Init payment: create pending DB record + return QR data ──────────
        if ($_POST['action'] === 'init_payment') {
            $plan_id = $_POST['plan_id'] ?? '';
            if (empty($plan_id)) throw new Exception('Please select a plan.');

            // Block duplicate active bookings
            $checkSub = $pdo->prepare("SELECT subscription_id FROM user_subscriptions WHERE user_id = ? AND subscription_status = 'Active' AND payment_status = 'Paid'");
            $checkSub->execute([$user_id]);
            if ($checkSub->fetch()) throw new Exception('You already have an active table booking.');

            $pStmt = $pdo->prepare("SELECT price, duration_days FROM subscription_plans WHERE plan_id = ? AND active = 1");
            $pStmt->execute([$plan_id]);
            $selectedPlan = $pStmt->fetch();
            if (!$selectedPlan) throw new Exception('Invalid plan selected.');

            $tStmt = $pdo->prepare("SELECT table_id, status FROM library_tables WHERE table_number = ?");
            $tStmt->execute([$table_id]);
            $tableRow = $tStmt->fetch();
            if (!$tableRow || $tableRow['status'] === 'Booked') throw new Exception('This table is no longer available.');

            $reference = 'TXN' . time() . mt_rand(1000, 9999);

            // Save pending records
            $pdo->beginTransaction();
            $pdo->prepare("INSERT INTO user_subscriptions (user_id, plan_id, table_id, start_date, expiry_date, amount_paid, payment_status, subscription_status) VALUES (?, ?, ?, CURDATE(), DATE_ADD(CURDATE(), INTERVAL ? DAY), ?, 'Pending', 'Active')")
                ->execute([$user_id, $plan_id, $tableRow['table_id'], $selectedPlan['duration_days'], $selectedPlan['price']]);
            $subscription_id = $pdo->lastInsertId();

            $pdo->prepare("INSERT INTO payments (user_id, subscription_id, amount, payment_method, payment_status, payment_reference, created_at, payment_date) VALUES (?, ?, ?, 'UPI', 'Pending', ?, NOW(), NOW())")
                ->execute([$user_id, $subscription_id, $selectedPlan['price'], $reference]);
            $pdo->commit();

            // Build UPI deep-link
            $upiUrl = 'upi://pay?' . http_build_query([
                'pa' => PAYMENT_UPI_ID,
                'pn' => PAYMENT_PAYEE_NAME,
                'am' => number_format($selectedPlan['price'], 2, '.', ''),
                'cu' => PAYMENT_CURRENCY,
                'tn' => 'Library table booking ' . $reference,
                'tr' => $reference,
            ]);

            jsonResponse([
                'ok'            => true,
                'reference'     => $reference,
                'amount'        => $selectedPlan['price'],
                'upi_url'       => $upiUrl,
                'rzp_link'      => RAZORPAY_PAYMENT_LINK,
                'window'        => PAYMENT_WINDOW_SECONDS,
                'upi_id'        => PAYMENT_UPI_ID,
                'payee_name'    => PAYMENT_PAYEE_NAME,
            ]);
        }

        // ── Check payment status (user clicked "I've Paid") ───────────────────
        if ($_POST['action'] === 'check_payment') {
            $reference = $_POST['reference'] ?? '';
            if (empty($reference)) throw new Exception('Payment reference is missing.');
            jsonResponse(['ok' => true] + completePaidBooking($pdo, $user_id, $table_id, $reference, $user));
        }
    } catch (Exception $e) {
        jsonResponse(['ok' => false, 'message' => $e->getMessage()]);
    }
}

        <script>
        var TABLE_ID = <?= json_encode($table_id) ?>;
        var POLL_MS = <?= PAYMENT_POLL_SECONDS * 1000 ?>;
        var WINDOW_SECS = <?= PAYMENT_WINDOW_SECONDS ?>;
        var CIRCUM = 2 * Math.PI * 30; // r=30

        var planSelect = document.getElementById('planSelect');
        var paymentBox = document.getElementById('paymentBox');
        var priceDisplay = document.getElementById('priceDisplay');
        var statusBox = document.getElementById('paymentStatus');
        var dynamicQrImg = document.getElementById('dynamicQrImg');
        var paymentActions = document.getElementById('paymentActions');
        var checkPaymentBtn = document.getElementById('checkPaymentBtn');
        var paidConfirm = document.getElementById('paidConfirm');
        var iPaidBtn = document.getElementById('iPaidBtn');
        var previewPlan = document.getElementById('previewPlan');
        var previewStatus = document.getElementById('previewStatus');
        var rzpLinkBtn = document.getElementById('rzpLinkBtn');
        var ringFill = document.getElementById('ringFill');
        var ringText = document.getElementById('ringText');

        var activeReference = '';
        var pollTimer = null;
        var countdownTimer = null;
        var windowStart = null;

        function showStatus(message, type) {
            if (typeof type === 'undefined') {
                type = 'waiting';
            }

            if (statusBox) {
                if (type === 'waiting') {
                    statusBox.innerHTML = "<span class=\"spinner\"></span>" + message;
                } else {
                    statusBox.innerHTML = message;
                }
                statusBox.className = 'payment-status status-' + type;
                statusBox.style.display = 'block';
            }

            if (previewStatus) {
                if (type === 'success') {
                    previewStatus.textContent = '✅ Confirmed';
                } else if (type === 'error') {
                    previewStatus.textContent = '❌ Failed';
                } else if (type === 'timeout') {
                    previewStatus.textContent = '⏰ Timed out';
                } else {
                    previewStatus.textContent = '⏳ Waiting';
                }
            }
        }

        function updateCountdown() {
            if (!windowStart) {
                return;
            }

            var elapsed = (Date.now() - windowStart) / 1000;
            var remaining = Math.max(0, WINDOW_SECS - elapsed);
            var mins = String(Math.floor(remaining / 60)).padStart(1, "0");
            var secs = String(Math.floor(remaining % 60)).padStart(2, "0");

            if (ringText) {
                ringText.textContent = mins + ":" + secs;
            }
            if (ringFill) {
                ringFill.style.strokeDashoffset = CIRCUM * (1 - remaining / WINDOW_SECS);
                if (remaining <= 60) {
                    ringFill.classList.add("urgent");
                } else {
                    ringFill.classList.remove("urgent");
                }
            }

            if (remaining <= 0) {
                clearInterval(countdownTimer);
                showStatus("⏰ Payment window expired. Click below to check if payment was received.", "timeout");
                pollPaymentStatus();
            }
        }

        function postAction(action, extra) {
            extra = extra || {};
            var bodyParts = [];
            bodyParts.push("action=" + encodeURIComponent(action));
            Object.keys(extra).forEach(function(key) {
                bodyParts.push(encodeURIComponent(key) + "=" + encodeURIComponent(extra[key]));
            });
            var body = bodyParts.join("&");

            return fetch("payment.php?table_id=" + encodeURIComponent(TABLE_ID), {
                method: "POST",
                headers: {
                    "Content-Type": "application/x-www-form-urlencoded"
                },
                body: body
            }).then(function(res) {
                return res.json();
            });
        }

        function pollPaymentStatus() {
            if (!activeReference) {
                return;
            }

            postAction("check_payment", {
                reference: activeReference
            }).then(function(result) {
                if (!result.ok) {
                    showStatus("⚠️ " + (result.message || "Could not verify payment."), "error");
                    return;
                }

                if (result.completed) {
                    clearInterval(pollTimer);
                    clearInterval(countdownTimer);
                    showStatus("✅ Payment confirmed! Redirecting…", "success");
                    setTimeout(function() {
                        window.location.href = result.redirect;
                    }, 1200);
                    return;
                }

                if (result.failed || result.timed_out) {
                    clearInterval(pollTimer);
                    clearInterval(countdownTimer);
                    showStatus("❌ " + (result.message || "Payment not completed."), "error");
                    setTimeout(function() {
                        window.location.href = result.redirect;
                    }, 2500);
                    return;
                }
            }).catch(function() {
                showStatus("⚠️ Connection issue. Will retry automatically.", "error");
            });
        }

        function createPaymentRequest() {
            var opt = planSelect.options[planSelect.selectedIndex];
            var price = opt ? opt.getAttribute('data-price') : null;
            var planId = planSelect ? planSelect.value : '';

            clearInterval(pollTimer);
            clearInterval(countdownTimer);
            activeReference = '';
            windowStart = null;

            if (paymentBox) paymentBox.style.display = 'none';
            if (statusBox) statusBox.style.display = 'none';
            if (paymentActions) paymentActions.style.display = 'none';
            if (paidConfirm) paidConfirm.style.display = 'none';
            if (previewPlan) previewPlan.textContent = '—';
            if (previewStatus) previewStatus.textContent = 'Select plan';
            if (!price || !planId) {
                return;
            }

            if (previewPlan && opt) previewPlan.textContent = opt.textContent.trim();
            if (priceDisplay) priceDisplay.textContent = '₹' + parseFloat(price).toLocaleString('en-IN');

            showStatus('Creating payment request…', 'waiting');
            if (paymentBox) paymentBox.style.display = 'block';
            if (paymentActions) paymentActions.style.display = 'flex';

            postAction('init_payment', {
                plan_id: planId
            }).then(function(result) {
                if (!result.ok) {
                    if (paymentBox) paymentBox.style.display = 'none';
                    showStatus('❌ ' + (result.message || 'Could not start payment.'), 'error');
                    return;
                }

                activeReference = result.reference;
                windowStart = Date.now();

                if (dynamicQrImg) {
                    dynamicQrImg.src = 'https://api.qrserver.com/v1/create-qr-code/?size=220x220&data=' +
                        encodeURIComponent(result.upi_url);
                }

                if (rzpLinkBtn) {
                    rzpLinkBtn.href = result.rzp_link;
                }

                if (ringFill) {
                    ringFill.style.strokeDashoffset = 0;
                    ringFill.classList.remove('urgent');
                }
                if (ringText) {
                    ringText.textContent = '5:00';
                }
                countdownTimer = setInterval(updateCountdown, 1000);

                if (paidConfirm) {
                    paidConfirm.style.display = 'block';
                }

                showStatus(
                    'Scan the QR above or click the Razorpay link to pay. After paying, click "I\'ve Paid".',
                    'waiting');
                if (previewStatus) previewStatus.textContent = '⏳ Awaiting payment';

                pollTimer = setInterval(pollPaymentStatus, POLL_MS);
            }).catch(function() {
                if (paymentBox) paymentBox.style.display = 'none';
                showStatus('❌ Could not create payment request. Please try again.', 'error');
            });
        }

        if (planSelect) {
            planSelect.addEventListener('change', createPaymentRequest);
        }

        if (iPaidBtn) {
            iPaidBtn.addEventListener('click', function() {
                iPaidBtn.disabled = true;
                iPaidBtn.textContent = 'Checking…';
                showStatus('Checking payment with server…', 'waiting');
                pollPaymentStatus();
                setTimeout(function() {
                    if (iPaidBtn && iPaidBtn.disabled) {
                        iPaidBtn.disabled = false;
                        iPaidBtn.textContent = '✅ I\'ve Paid — Confirm My Booking';
                    }
                }, 4000);
            });
        }

        if (checkPaymentBtn) {
            checkPaymentBtn.addEventListener('click', function() {
                showStatus('Checking payment status…', 'waiting');
                pollPaymentStatus();
            });
        }
        </script>
